1 cpp 源码

#include <iostream>
// 读入 n 个数求和
int main() {
    int n = 3;
    double sum = 0;
    std::cout << "n = " << n << std::endl;
    for (int i = 0; i < n; ++i) {
        sum += i * 1.5;
    }
    return 0;
}
